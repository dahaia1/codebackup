import pandas as pd
#解决数据输出时列不对齐的问题
pd.set_option('display.unicode.east_asian_width', True)
df = pd.DataFrame({
    '姓名':['甲','乙','丙'],
    '语文':[110,105,109],
     '数学':[105,88,120],
     '英语':[99,115,130]})
# 设置"姓名"为索引
df=df.set_index('姓名')
print(df)
# 重新设置行索引
df_row=df.reindex(['甲','乙','丙','丁','戊'])
print(df_row)
# 重新设置列索引
df_col=df.reindex(columns=['语文','物理','数学','英语'])
print(df_col)
df=df.reindex(index=['甲','乙','丙','丁','戊'],columns=['语文','物理','数学','英语'])
print(df)
